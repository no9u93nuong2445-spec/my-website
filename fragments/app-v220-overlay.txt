  function getRealScenarioV220(currentSession = session) {
    const id = currentSession?.realScenario?.id;
    return id ? REAL_SCENARIOS_V220[id] || null : null;
  }

  function buildRealScenarioLevelV220(scene, char) {
    const config = normalizeFreeConfig(scene.config);
    return {
      id: `real-${scene.id}`, chapter: 0, no: "实战", title: scene.title, character: char.id,
      intro: scene.intro, objective: scene.objective, opening: scene.opening,
      minTurns: scene.minTurns, maxTurns: scene.maxTurns, targetAvg: scene.targetAvg,
      required: {}, hints: scene.hints, references: scene.references,
      replies: {
        positive: ["这样说我比较容易接受。", "嗯，谢谢你把话说清楚。"],
        neutral: ["嗯。", "我知道了。"],
        negative: ["我不太想继续解释。", "这样会让我有压力。"],
        boundary: ["我已经把想法说清楚了，先到这里吧。"]
      },
      freeConfig: config
    };
  }

  function startRealScenarioV220(sceneId) {
    const scene = REAL_SCENARIOS_V220[sceneId];
    if (!scene) return false;
    const char = CHARACTERS[scene.characterId];
    const level = buildRealScenarioLevelV220(scene, char);
    const config = normalizeFreeConfig(scene.config);
    state.game.freePractice.lastConfig = { ...config };
    clearActiveSession();
    saveState();
    session = createSession(level, char, true, config);
    session.realScenario = { id: scene.id, rewarded: false, safeTurns: 0, boundaryTurns: 0 };
    session.metrics.comfort = clamp(session.metrics.comfort + scene.metrics.comfort, 8, 90);
    session.metrics.interest = clamp(session.metrics.interest + scene.metrics.interest, 4, char.maxInterest || 90);
    session.metrics.pressure = clamp(session.metrics.pressure + scene.metrics.pressure, 0, 88);
    persistActiveSession(session);
    renderTraining();
    return true;
  }

  function getRealScenarioStatusV220(currentSession) {
    const scene = getRealScenarioV220(currentSession);
    if (!scene) return null;
    const analyses = currentSession.analyses || [];
    const has = key => analyses.some(a => a.flags?.[key]);
    const hasIssue = key => analyses.some(a => a.issues?.some(i => i.key === key));
    const safe = !has("pressure") && !has("insult") && !has("sexual") && !has("mindReading") && !has("scorekeeping");
    let met = false;
    let progress = "";
    if (scene.id === "dryHaha") {
      met = currentSession.turn >= 2 && currentSession.counters.share >= 1 && currentSession.counters.relevant >= 1 && !hasIssue("interrogation") && safe;
      progress = `交流 ${currentSession.turn}/2轮 · 分享 ${currentSession.counters.share}/1 · 承接 ${currentSession.counters.relevant}/1 · ${hasIssue("interrogation") ? "出现密集提问" : "提问节奏正常"}`;
    } else if (scene.id === "cooling") {
      const space = currentSession.counters.acceptUncertainty + currentSession.counters.ending;
      met = currentSession.turn >= 2 && space >= 1 && safe;
      progress = `交流 ${currentSession.turn}/2轮 · 给空间/收尾 ${space}/1 · ${has("pressure") ? "出现催促" : "没有催回复"}`;
    } else if (scene.id === "inviteRejected") {
      met = currentSession.turn >= 2 && currentSession.counters.respectRejection >= 1 && currentSession.counters.ending >= 1 && safe;
      progress = `接受拒绝 ${currentSession.counters.respectRejection}/1 · 自然收尾 ${currentSession.counters.ending}/1 · ${has("invite") ? "仍有继续邀约" : "没有绕过拒绝"}`;
    } else if (scene.id === "afterDate") {
      const warm = currentSession.counters.specificCompliment + currentSession.counters.share + currentSession.counters.ending;
      met = currentSession.turn >= 2 && currentSession.counters.relevant >= 1 && warm >= 2 && !has("intimacy") && !has("mindReading") && safe;
      progress = `交流 ${currentSession.turn}/2轮 · 具体回应/分享/收尾 ${warm}/2 · ${has("intimacy") ? "推进过快" : "没有逼定义关系"}`;
    } else if (scene.id === "relationshipCooling") {
      const selfCare = currentSession.counters.selfBoundary + currentSession.counters.ending;
      met = currentSession.turn >= 2 && currentSession.counters.respectRejection >= 1 && selfCare >= 1 && safe;
      progress = `接受边界 ${currentSession.counters.respectRejection}/1 · 自我边界/收尾 ${selfCare}/1 · ${safe ? "没有继续施压" : "仍存在推进或脑补"}`;
    }
    return { id: scene.id, title: scene.title, met, progress, objective: scene.objective };
  }

  function addScenarioIssueV220(analysis, text) {
    if (!analysis.issues.some(item => item.key === "scenarioPace")) analysis.issues.push({ key: "scenarioPace", text });
  }

  function applyScenarioScoreV220(analysis, currentSession) {
    const scene = getRealScenarioV220(currentSession);
    if (!scene) return analysis;
    const f = analysis.flags;
    let delta = 0;
    const praise = text => { if (!analysis.strengths.includes(text)) analysis.strengths.push(text); };
    if (scene.id === "dryHaha") {
      if ((f.share || f.conversationRepair || analysis.relevant) && analysis.questionCount <= 1) { delta += 7; praise("没有因为短回复慌张，而是提供了新的可聊内容。"); }
      if (analysis.questionCount >= 2) { delta -= 10; addScenarioIssueV220(analysis, "面对短回复连续提问，会让救场变成新的压力。"); }
      if (f.pressure || f.intimacy || f.mindReading) { delta -= 14; addScenarioIssueV220(analysis, "不能把一句‘哈哈’解释成喜欢、敷衍或要求对方解释。"); }
    } else if (scene.id === "cooling") {
      if (f.acceptUncertainty || f.ending) { delta += 10; praise("接受了她当前的回复节奏，没有要求即时证明态度。"); }
      if (f.pressure || f.mindReading) { delta -= 18; addScenarioIssueV220(analysis, "她已经说明忙碌，再追问回复速度只会增加压力。"); }
      if (analysis.questionCount >= 2) delta -= 6;
    } else if (scene.id === "inviteRejected") {
      if (f.respectRejection) { delta += 18; praise("明确接受了这次拒绝，没有换一种说法继续争取。"); }
      if (f.ending) delta += 7;
      if (f.invite || f.intimacy || f.mindReading) { delta -= 20; addScenarioIssueV220(analysis, "继续换时间、猜原因或表达更重感情，都在绕过她的拒绝。"); }
    } else if (scene.id === "afterDate") {
      if (f.specificCompliment) { delta += 8; praise("把好感落到了具体相处细节，没有空泛吹捧。"); }
      if (f.share || analysis.relevant || f.ending) delta += 5;
      if (currentSession.turn <= 1 && (f.interestSignal || f.intimacy || f.invite || f.mindReading)) { delta -= 12; addScenarioIssueV220(analysis, "第一次见面后立刻确认关系、评价或下一次安排，节奏偏快。"); }
      if (analysis.questionCount >= 2) delta -= 5;
    } else if (scene.id === "relationshipCooling") {
      if (f.respectRejection) { delta += 16; praise("听懂了‘减少联系’的明确边界，没有继续证明自己。"); }
      if (f.selfBoundary) { delta += 10; praise("尊重对方的同时，也照顾了自己的时间和投入。"); }
      if (f.ending) delta += 7;
      if (f.pressure || f.invite || f.mindReading || f.scorekeeping || f.intimacy) { delta -= 22; addScenarioIssueV220(analysis, "关系已经明确降温，继续追问、证明或索取回报会越过边界。"); }
    }
    analysis.score = clamp(Math.round(analysis.score + delta), 0, 100);
    analysis.scenario = { id: scene.id, title: scene.title, delta };
    if (delta > 0 && !analysis.intents.includes("处理真实场景")) analysis.intents.push("处理真实场景");
    return analysis;
  }

  generateResponse = function generateResponseV220(analysis, currentSession) {
    const base = generateResponseV219ScenarioBase(analysis, currentSession);
    const scene = getRealScenarioV220(currentSession);
    if (!scene) return base;
    const f = analysis.flags;
    let text = base.text;
    let reason = "她会根据你是否尊重当前场景中的事实和边界继续回应。";
    if (scene.id === "dryHaha") {
      if (f.pressure || f.mindReading || analysis.questionCount >= 2) { text = "你不用一次问这么多，我刚才就是顺手回了一下。"; reason = "你把短回复变成了追问压力，她会收紧交流。"; }
      else if (f.share || f.conversationRepair || analysis.relevant) { text = currentSession.turn >= 2 ? "这个我还能接，你继续说。" : "哈哈，这个确实有点画面感。你后来怎么处理的？"; reason = "你没有审问她，而是给出了新的内容，她愿意重新投入一点。"; }
      else { text = "哈哈，是有点。"; reason = "这句话仍缺少新内容，她只会保持短回复。"; }
    } else if (scene.id === "cooling") {
      if (f.pressure || f.mindReading) { text = "我已经说最近比较忙了，不想因为回复速度一直解释。"; reason = "她已说明现实原因，继续催促会触发边界。"; }
      else if (f.acceptUncertainty || f.ending) { text = "好，谢谢你理解。我有空会正常回。"; reason = "你接受了当前节奏，她不需要继续防御。"; }
      else if (f.share || analysis.relevant) { text = "嗯，我最近确实事情比较多，状态稳定点再聊。"; reason = "你接住了现实情况，但她仍会保持较低投入。"; }
      else { text = "嗯，最近就是有点忙。"; }
    } else if (scene.id === "inviteRejected") {
      if (f.invite || f.mindReading || f.intimacy) { text = "我这次不是换个时间的问题，最近确实不想约。"; reason = "继续换时间或猜原因，等于绕过她刚才的拒绝。"; }
      else if (f.respectRejection) { text = "谢谢你理解，先按现在的节奏就好。"; reason = "你明确接受拒绝，她不需要再次解释。"; }
      else if (f.ending) { text = "好，那先这样。"; reason = "你停止推进，场景可以自然结束。"; }
      else { text = "嗯，我最近确实不太想安排见面。"; }
    } else if (scene.id === "afterDate") {
      if (currentSession.turn <= 2 && (f.intimacy || f.interestSignal || f.invite || f.mindReading)) { text = "今天相处还可以，不过先不用这么快定下一次。"; reason = "一次见面后推进过快，她会主动放慢节奏。"; }
      else if (f.specificCompliment || f.share || analysis.relevant) { text = "我也觉得今天挺轻松的，路上那段还蛮好笑。"; reason = "你回应了具体经历，她可以自然确认这次相处感受。"; }
      else if (f.ending) { text = "好，你也早点休息。"; reason = "你没有趁热追问，收尾比较轻松。"; }
      else { text = "嗯，我已经到家了。"; }
    } else if (scene.id === "relationshipCooling") {
      if (f.pressure || f.invite || f.mindReading || f.scorekeeping || f.intimacy) { text = "我说想减少联系，不是让你继续证明或追问。"; reason = "她的边界被绕过，会再次明确降温。"; }
      else if (f.respectRejection || f.selfBoundary) { text = "谢谢你理解。我们都先把注意力放回自己的生活吧。"; reason = "你接受现实并照顾自己，双方不需要继续拉扯。"; }
      else if (f.ending) { text = "好，先这样。"; reason = "你停止推进，边界得到尊重。"; }
      else { text = "我现在确实需要一点距离。"; }
    }
    return { ...base, text, reason, state: getVisibleCharacterState(currentSession), scenario: scene.id };
  };

  const evaluateSessionV219ScenarioBase = evaluateSession;
  evaluateSession = function evaluateSessionV220(currentSession, forceFinish = false) {
    const status = getRealScenarioStatusV220(currentSession);
    if (!status) return evaluateSessionV219ScenarioBase(currentSession, forceFinish);
    const analyses = currentSession.analyses || [];
    const avg = analyses.length ? Math.round(analyses.reduce((sum, item) => sum + item.score, 0) / analyses.length) : 0;
    const severe = analyses.some(item => item.flags.insult || item.flags.sexual) || currentSession.metrics.pressure >= 92;
    const target = currentSession.level.targetAvg + (state.settings.strictMode ? 5 : 0);
    const enough = currentSession.turn >= currentSession.level.minTurns;
    const passed = !severe && status.met && enough && avg >= target;
    if (severe) return { finished: true, passed: false, avg, goalMet: false, scenarioStatus: status, reason: "场景中出现明显越界或对方压力过高。" };
    if (passed) return { finished: true, passed: true, avg, goalMet: true, scenarioStatus: status, reason: `已完成真实场景“${status.title}”。` };
    if (forceFinish || currentSession.turn >= currentSession.level.maxTurns) return { finished: true, passed: false, avg, goalMet: status.met, scenarioStatus: status, reason: status.met ? `关键动作已完成，但平均分尚未达到${target}分。` : `尚未完成真实场景“${status.title}”的关键动作。` };
    return { finished: false, passed: false, avg, goalMet: status.met, scenarioStatus: status, reason: "" };
  };

  function exportData() {
    const envelope = buildExportEnvelope(state);
    const blob = new Blob([JSON.stringify(envelope, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement("a");
    anchor.href = url;
    anchor.download = `心动训练营_完整备份_V${APP_VERSION}_${new Date().toISOString().slice(0,10)}.json`;
    document.body.appendChild(anchor);
    anchor.click();
    anchor.remove();
    URL.revokeObjectURL(url);
  }

  async function importDataFile(file) {
    if (!file) return false;
    if (Number(file.size) > MAX_IMPORT_BYTES) throw new Error("导入文件超过5MB，已停止处理。");
    const text = await file.text();
    const parsed = JSON.parse(text);
    const imported = parseImportedPayload(parsed);
    const summary = getStateSummary(imported);
    const confirmed = confirm(`准备恢复这份数据：\n已通关 ${summary.completed}/${TOTAL_LEVELS} 关\n经验 ${summary.xp} XP\n历史记录 ${summary.history} 局\n课程 ${summary.courses}/12\n\n当前数据会先自动备份。确认继续吗？`);
    if (!confirmed) return false;
    createLocalBackup("导入前备份", state, true);
    replaceState(imported);
    saveState({ skipBackup: true });
    applyAppearance();
    startupDataNotice = "数据导入成功；导入前的存档已保存在本地备份中。";
    currentView = "settings";
    renderSettings();
    alert("数据恢复完成。");
    return true;
  }

  function restoreLocalBackup(backupId) {
    const backup = readLocalBackups().find(item => item.id === backupId && isBackupValid(item));
    if (!backup) { alert("这份备份不存在或校验失败。"); return false; }
    const summary = getStateSummary(backup.state);
    if (!confirm(`恢复“${backup.reason}”？\n${formatHistoryDate(backup.createdAt)}\n已通关 ${summary.completed}/${TOTAL_LEVELS} 关，${summary.xp} XP。\n\n当前数据会先保存为一份“恢复前备份”。`)) return false;
    createLocalBackup("恢复前备份", state, true);
    replaceState(backup.state);
    saveState({ skipBackup: true });
    applyAppearance();
    startupDataNotice = `已恢复备份：${backup.reason}`;
    renderSettings();
    return true;
  }

  function deleteTrainingHistory(historyId) {
    const before = state.game.trainingHistory.length;
    if (!historyId || !state.game.trainingHistory.some(item => item.id === historyId)) return false;
    createLocalBackup("删除训练记录前", state, true);
    state.game.trainingHistory = state.game.trainingHistory.filter(item => item.id !== historyId);
    saveState({ skipBackup: true });
    return state.game.trainingHistory.length === before - 1;
  }

  function formatBackupSummary(backup) {
    try {
      const summary = getStateSummary(backup.state);
      return `${summary.completed}/${TOTAL_LEVELS}关 · ${summary.xp} XP · ${summary.history}局历史`;
    } catch {
      return "备份摘要不可读取";
    }
  }

  function showLockScreen(message = "") {
    if (!lockConfig.enabled) return;
    appUnlocked = false;
    document.body?.classList?.add("app-locked");
    if (unlockError) unlockError.textContent = message;
    if (unlockPassword) unlockPassword.value = "";
    try { if (!lockDialog?.open) lockDialog?.showModal?.(); } catch {}
    setTimeout(() => unlockPassword?.focus?.(), 30);
  }

  function hideLockScreen() {
    appUnlocked = true;
    failedUnlockAttempts = 0;
    document.body?.classList?.remove("app-locked");
    try { lockDialog?.close?.(); } catch {}
    if (!state.settings.onboardingDone) openOnboarding(false);
  }

  async function handleUnlock() {
    const password = unlockPassword?.value || "";
    if (!password) { if (unlockError) unlockError.textContent = "请输入本地密码。"; return; }
    const ok = await verifyLocalPassword(password);
    if (ok) { hideLockScreen(); return; }
    failedUnlockAttempts += 1;
    if (unlockError) unlockError.textContent = failedUnlockAttempts >= 3 ? "密码不正确。忘记密码只能清除本浏览器的站点数据，训练数据也会被删除。" : "密码不正确，请重试。";
    if (unlockPassword) { unlockPassword.value = ""; unlockPassword.focus?.(); }
  }

  function startLevel(levelId) {
    const level = LEVELS.find(item => item.id === levelId);

    const char = CHARACTERS[level.character];
    clearActiveSession();
    session = createSession(level, char, false);
    persistActiveSession(session);
    renderTraining();
  }

  function normalizeFreeConfig(input, legacyScene = null) {
    let config;
    if (typeof input === "string") {
      const legacy = {
        opening: { characterId: input, channel: "social", stage: "new", enthusiasm: "normal", mood: "relaxed", difficulty: "beginner", goal: "balanced" },
        daily: { characterId: input, channel: "wechat", stage: "few", enthusiasm: "normal", mood: "busy", difficulty: "standard", goal: "balanced" },
        cold: { characterId: input, channel: "social", stage: "few", enthusiasm: "cold", mood: "cautious", difficulty: "standard", goal: "boundary" },
        invite: { characterId: input, channel: "wechat", stage: "familiar", enthusiasm: "warm", mood: "relaxed", difficulty: "standard", goal: "invite" }
      };
      config = legacy[legacyScene] || legacy.opening;
    } else {
      config = { ...(input || {}) };
    }
    const valid = (group, value, fallback) => Object.prototype.hasOwnProperty.call(group, value) ? value : fallback;
    return {
      characterId: Object.prototype.hasOwnProperty.call(CHARACTERS, config.characterId) ? config.characterId : "linxia",
      channel: valid(FREE_PRACTICE_OPTIONS.channels, config.channel, "social"),
      stage: valid(FREE_PRACTICE_OPTIONS.stages, config.stage, "new"),
      enthusiasm: valid(FREE_PRACTICE_OPTIONS.enthusiasm, config.enthusiasm, "normal"),
      mood: valid(FREE_PRACTICE_OPTIONS.moods, config.mood, "relaxed"),
      difficulty: valid(FREE_PRACTICE_OPTIONS.difficulties, config.difficulty, "beginner"),
      goal: valid(FREE_PRACTICE_OPTIONS.goals, config.goal, "balanced")
    };
  }

  function describeFreeConfig(config) {
    const c = normalizeFreeConfig(config);
    return `${FREE_PRACTICE_OPTIONS.channels[c.channel].label} · ${FREE_PRACTICE_OPTIONS.stages[c.stage].label} · ${FREE_PRACTICE_OPTIONS.enthusiasm[c.enthusiasm].label} · ${FREE_PRACTICE_OPTIONS.moods[c.mood].label} · ${FREE_PRACTICE_OPTIONS.difficulties[c.difficulty].label}`;
  }

  function stableIndex(text, length) {
    let hash = 0;
    for (const char of String(text)) hash = ((hash << 5) - hash + char.charCodeAt(0)) | 0;
    return length ? Math.abs(hash) % length : 0;
  }

  function buildFreeOpening(config, char) {
    const moodPool = FREE_OPENING_POOLS[config.mood]?.[char.id] || FREE_OPENING_POOLS.relaxed[char.id];
    let opening = moodPool[stableIndex(`${char.id}-${config.channel}-${config.stage}-${config.enthusiasm}`, moodPool.length)];
    if (config.enthusiasm === "cold") {

      <div class="feedback-section"><strong>下一步</strong><p>${escapeHtml(feedback.nextStep)}</p></div>`;
  }

  function showHint() {
    const hints = session.level.hints || [];
    session.usedHint = true;
    state.game.totalHints += 1;
    saveState();
    if (!hints.length) return;
    const hint = hints[session.hintIndex % hints.length];
    session.hintIndex += 1;
    session.lastFeedback = {
      score: 0, grade: "提示", intents: ["训练提示"],
      strengths: [hint], issues: [],
      feeling: "提示不会代替你作答，尽量用自己的语言。",
      reference: "先回应她上一句话，再加入一点自己的内容。",
      nextStep: "组织好以后再发送。"
    };
    persistActiveSession(session);
    renderTraining();
  }

  function showReference() {
    const refs = session.level.references || [];
    session.usedReference = true;
    state.game.totalReferences += 1;
    saveState();
    const ref = refs[Math.min(session.turn, refs.length - 1)] || refs[0] || "先回应，再分享，最后留下容易回答的空间。";
    session.lastFeedback = {
      score: 0, grade: "参考思路", intents: ["参考方向"],
      strengths: ["这是方向示例，不是唯一正确答案。"], issues: [],
      feeling: "照抄不等于掌握，最好改成自己的语气。",
      reference: ref,
      nextStep: "用自己的表达重新组织后再发送。"
    };
    persistActiveSession(session);
    renderTraining();
  }

  function finishFreePractice() {
    if (!session?.isFree || session.finished) return;
    if (session.turn < 1) {
      alert("至少先发送一句话，系统才能生成有效复盘。");
      return;
    }
    if (session.turn < session.level.minTurns && !confirm(`目前只练了${session.turn}轮，提前结束可能无法达成目标。仍然生成复盘吗？`)) return;
    const result = evaluateSession(session, true);
    session.finished = true;
    finalizeSession(result);
  }

  function sendMessage() {
    if (!session || session.finished) return;
    const input = document.getElementById("messageInput");
    const text = input.value.trim();
    if (!text) {
      input.focus();
      return;
    }

    session.inputDraft = "";
    const analysis = analyzeMessage(text, session);
    session.turn += 1;
    session.analyses.push(analysis);
    session.messages.push({ from: "me", text, score: analysis.score });
    updateSessionMetrics(analysis);
    updateConversationMemory(analysis, session);
    updateGlobalStats(analysis, session);

    const response = generateResponse(analysis, session);
    session.messages.push({ from: "her", text: response.text });
    if (/[？?]/.test(response.text)) session.counters.herQuestions = (session.counters.herQuestions || 0) + 1;
    rememberHerResponse(response.text, session);
    session.lastFeedback = buildFeedback(analysis, session, response);

    const result = evaluateSession(session);
    if (result.finished) {
      session.finished = true;
      finalizeSession(result);
      return;
    }

    persistActiveSession(session);
    renderTraining();
  }

    const acknowledges = contains([/原来/, /难怪/, /确实/, /听起来/, /看来/, /那还挺/, /这样的话/, /懂了/, /可以理解/]);
    const profileStarter = flags.profileReference && currentSession.level.chapter <= 2;
    const relevant = topicMatch.matched || acknowledges || flags.empathy || flags.ending || flags.respectRejection || flags.invite || flags.conversationRepair || profileStarter || (questionCount > 0 && refersToLastMessage(normalized));
    const openQuestion = questionCount === 1 && contains([/你觉得/, /你最/, /哪一个?/, /是什么/, /比较喜欢/, /更喜欢/, /感觉怎么样/, /怎么会/, /哪部分/, /还是/, /平时会/, /后来呢/, /为什么会/]);

    }

    } else if (flags.compliment && !flags.overPraise) {

      addIssue("pressure", "这句话带有催促、质问或要求对方立刻回应的感觉。");

      score += currentSession.char.weights.intimacy || -8;
      score -= 13;
      addIssue("overIntimacy", "当前关系还没到这个亲密程度。");

      addIssue("adviceRush", "对方还没邀请你给方案，直接说教容易让人不舒服。");
    }
    if (flags.overPraise) {

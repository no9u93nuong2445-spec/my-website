  function normalizeStateData(parsed) {
    const base = defaultState();
    const source = parsed && typeof parsed === "object" ? parsed : {};
    const merged = {
      ...base,
      ...source,
      meta: { ...base.meta, ...(source.meta || {}) },
      stats: { ...base.stats, ...(source.stats || {}) },
      game: {
        ...base.game,
        ...(source.game || {}),
        achievements: { ...base.game.achievements, ...(source.game?.achievements || {}) },
        chapterAwards: { ...base.game.chapterAwards, ...(source.game?.chapterAwards || {}) },
        mistakeMastery: { ...base.game.mistakeMastery, ...(source.game?.mistakeMastery || {}) },
        trainingHistory: Array.isArray(source.game?.trainingHistory) ? source.game.trainingHistory.slice(-120) : [],
        freePractice: { ...base.game.freePractice, ...(source.game?.freePractice || {}) },
        learning: {
          ...base.game.learning,
          ...(source.game?.learning || {}),
          completed: { ...base.game.learning.completed, ...(source.game?.learning?.completed || {}) },
          attempts: { ...base.game.learning.attempts, ...(source.game?.learning?.attempts || {}) }
        },
        personalization: { ...base.game.personalization, ...(source.game?.personalization || {}) }
      },
      settings: { ...base.settings, ...(source.settings || {}) }
    };
    if (merged.levels?.[15]?.completed && merged.unlocked < 16) merged.unlocked = 16;
    const completedIds = Object.entries(merged.levels || {}).filter(([, record]) => record?.completed).map(([id]) => Number(id)).filter(Number.isFinite);
    const highestCompleted = completedIds.length ? Math.max(...completedIds) : 0;
    if (highestCompleted >= Number(merged.unlocked || 1) && highestCompleted < TOTAL_LEVELS) merged.unlocked = highestCompleted + 1;
    if (merged.game.achievements?.allLevels && highestCompleted >= 30 && highestCompleted < TOTAL_LEVELS) {
      merged.game.achievements.foundationGraduate ||= merged.game.achievements.allLevels;
      delete merged.game.achievements.allLevels;
    }
    merged.unlocked = clamp(Number(merged.unlocked) || 1, 1, TOTAL_LEVELS);
    merged.game.xp = Math.max(0, Number(merged.game.xp) || 0);
    merged.game.trainingDays = Array.isArray(merged.game.trainingDays) ? merged.game.trainingDays.slice(-90) : [];
    merged.game.activeSession = normalizeSessionDraft(merged.game.activeSession);
    merged.settings.theme = ["system", "light", "dark"].includes(merged.settings.theme) ? merged.settings.theme : "system";
    merged.settings.fontSize = ["small", "standard", "large"].includes(merged.settings.fontSize) ? merged.settings.fontSize : "standard";
    merged.settings.reduceMotion = Boolean(merged.settings.reduceMotion);
    merged.settings.onboardingDone = Boolean(merged.settings.onboardingDone);
    merged.settings.autoBackup = merged.settings.autoBackup !== false;
    merged.meta.schemaVersion = DATA_SCHEMA_VERSION;

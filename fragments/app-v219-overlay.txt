  function ensureV219Session(s) {
    if (!s) return null;
    s.personality ||= ROLE_PERSONALITIES_V219[s.char?.id] || ROLE_PERSONALITIES_V219.linxia;
    s.trainingMode ||= state.settings.coachMode === false ? "immersive" : "coach";
    if (!s.isFree && !s.isMistake && !s.hiddenGoal) {
      const chapter = Number(s.level?.chapter || 1);
      const pools = chapter <= 1 ? ["listen","share","oneQuestion"] : chapter <= 2 ? ["balanced","listen","share"] : chapter <= 3 ? ["empathy","balanced","naturalClose"] : chapter <= 4 ? ["invite","naturalClose","oneQuestion"] : ["boundary","selfBoundary","naturalClose"];

    }
    return s;
  }

  function classifySemanticV219(a,text) {
    const f = a.flags || {};
    const topics = new Set([...(a.context?.topicMatch || []), ...detectTopicsFromText(text)]);
    if (f.invite) topics.add("meetup");

    if (f.respectRejection || f.interestSignal || f.mindReading) topics.add("relationship");
    const topicList = [...topics];
    const actions = [];
    const addAction = label => { if (label && !actions.includes(label)) actions.push(label); };
    if (f.respectRejection) addAction("尊重拒绝");
    if (f.ending) addAction("自然收尾");
    if (f.invite) addAction("发起邀约");
    if (f.empathy) addAction("回应情绪");
    if (f.apology) addAction("承担责任");
    if (f.conversationRepair) addAction("重新找话题");
    if (f.humor) addAction("轻松调侃");
    if (f.compliment) addAction("表达欣赏");
    if (f.share) addAction("自我分享");
    if (a.questionCount > 0) addAction("提问了解");

    const action = actions.join("＋");
    let object = topicList.length ? topicList.map(t => TOPIC_LABELS[t] || ({ relationship: "关系与边界", boundary: "个人边界", intimacy: "亲密推进", humor: "轻松幽默" }[t] || t)).join("、") : "当前交流";

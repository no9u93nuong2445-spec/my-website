      opening = opening.split(/[，。！？]/)[0] + "。";
    } else if (config.enthusiasm === "warm") {
      const followups = FREE_WARM_FOLLOWUPS[char.id];
      opening = `${opening}${followups[stableIndex(`${config.channel}-${config.goal}`, followups.length)]}`;
    }
    if (config.goal === "boundary" && !/慢|安静|不太想|不习惯|忙/.test(opening)) {
      opening = char.id === "linxia" ? "我今天状态一般，可能不会聊很多。" : char.id === "suyan" ? "我现在比较想安静一点，回复可能会短。" : "我今天比较忙，聊天频率不会很高。";
    }
    return opening;
  }

  function buildFreeLevel(config, char) {
    const channel = FREE_PRACTICE_OPTIONS.channels[config.channel];
    const stage = FREE_PRACTICE_OPTIONS.stages[config.stage];
    const goal = FREE_PRACTICE_OPTIONS.goals[config.goal];
    const difficulty = FREE_PRACTICE_OPTIONS.difficulties[config.difficulty];
    return {
      id: "free-custom", chapter: 0, no: "自由", title: `自由练习：${goal.label}`,
      character: char.id,
      intro: `${channel.intro}，${stage.intro}。她当前${FREE_PRACTICE_OPTIONS.moods[config.mood].label}，互动热情度为“${FREE_PRACTICE_OPTIONS.enthusiasm[config.enthusiasm].label}”。`,
      objective: goal.objective,
      opening: buildFreeOpening(config, char), minTurns: difficulty.minTurns, maxTurns: difficulty.maxTurns, targetAvg: difficulty.targetAvg,
      required: {},
      hints: ["先接住她上一句话，不要急着证明自己。", "优先使用‘回应＋一点自己＋一个可选问题’的结构。", goal.desc],
      references: ["先回应她刚才的具体内容，再补充一点与你有关的信息，最后只留一个容易回答的空间。"],
      replies: {
        positive: ["这个我愿意继续聊。", "嗯，这样说比较自然。", "你这个角度还挺好接。"],
        neutral: ["嗯。", "还行。", "是吗。"],
        negative: ["你问题有点多。", "不用这么着急。", "这个说法我不太好接。"],
        boundary: ["我现在不想继续这个话题。", "先到这里吧，我需要一点空间。"]
      },
      freeConfig: config
    };
  }

  function startFreeSession(configOrCharacter, legacyScene = null) {
    const config = normalizeFreeConfig(configOrCharacter, legacyScene);
    const char = CHARACTERS[config.characterId];
    const level = buildFreeLevel(config, char);
    state.game.freePractice.lastConfig = { ...config };
    clearActiveSession();
    saveState();
    session = createSession(level, char, true, config);
    persistActiveSession(session);
    renderTraining();
  }

  function getFreeMetricAdjustments(config) {
    const stage = FREE_PRACTICE_OPTIONS.stages[config.stage];
    const enthusiasm = FREE_PRACTICE_OPTIONS.enthusiasm[config.enthusiasm];
    const mood = FREE_PRACTICE_OPTIONS.moods[config.mood];
    const difficulty = FREE_PRACTICE_OPTIONS.difficulties[config.difficulty];
    return {
      comfort: stage.comfort + enthusiasm.comfort + mood.comfort + difficulty.comfort,
      interest: stage.interest + enthusiasm.interest + mood.interest + difficulty.interest,
      pressure: stage.pressure + enthusiasm.pressure + mood.pressure + difficulty.pressure,
      followup: enthusiasm.followup + mood.followup + difficulty.followup
    };
  }

  function createSession(level, char, isFree, freeConfig = null) {
    const openingTopics = detectTopicsFromText(`${level.intro || ""} ${level.opening || ""}`);
    const normalizedFree = isFree ? normalizeFreeConfig(freeConfig || level.freeConfig || {}) : null;
    const adjustment = normalizedFree ? getFreeMetricAdjustments(normalizedFree) : { comfort: 0, interest: 0, pressure: 0, followup: 0 };
    const topicCounts = {};
    openingTopics.forEach(topic => { topicCounts[topic] = (topicCounts[topic] || 0) + 1; });
    return {
      level,
      char,
      isFree,
      freeConfig: normalizedFree,
      freeResponseAdjustment: adjustment.followup,
      isMistake: false,
      focusIssue: null,
      baseLevelId: null,
      usedHint: false,
      usedReference: false,
      sessionXp: 0,
      newAchievements: [],
      messages: [{ from: "her", text: level.opening }],
      turn: 0,
      analyses: [],
      metrics: {
        comfort: clamp(char.initial.comfort + adjustment.comfort, 12, 82),
        interest: clamp(char.initial.interest + adjustment.interest, 8, Math.min(82, char.maxInterest || 82)),
        pressure: clamp(char.initial.pressure + adjustment.pressure, 0, 72),
        respect: 50
      },
      counters: {
        share: 0, relevant: 0, empathy: 0, humor: 0, ending: 0, invite: 0,
        concreteInvite: 0, noPressure: 0, openQuestion: 0, honestCuriosity: 0,
        specificCompliment: 0, noAdviceRush: 0, acceptUncertainty: 0,
        respectRejection: 0, interestSignal: 0, balanced: 0, respect: 0, herQuestions: 0,
        apology: 0, selfBoundary: 0, respectfulDifference: 0
      },
      memory: {
        recentTopics: openingTopics.slice(0, 4),
        topicCounts,
        topicHistory: [...openingTopics],
        userFacts: [],
        usedReplies: [],
        replyKeys: {},
        positiveStreak: 0,
        negativeStreak: 0,
        lastReaction: "保持观察"
      },
      lastFeedback: null,
      finished: false,
      hintIndex: 0,
      inputDraft: ""
    };
  }

  function ensureComposerVisible(smooth = true) {
    const area = document.querySelector?.(".input-area");

  }

  function getFreeConfigTags(config) {
    if (!config) return [];
    return [
      FREE_PRACTICE_OPTIONS.channels[config.channel].label,

      FREE_PRACTICE_OPTIONS.enthusiasm[config.enthusiasm].label,

      FREE_PRACTICE_OPTIONS.difficulties[config.difficulty].label
    ];
  }

  function getFreeGoalStatus(currentSession) {
    const goal = currentSession.freeConfig?.goal || "balanced";
    const counters = currentSession.counters;
    const topicMax = Math.max(0, ...Object.values(currentSession.memory.topicCounts || {}));
    const checks = {
      balanced: { met: counters.relevant >= 2 && counters.share >= 2 && counters.balanced >= 1, current: `${counters.relevant}/2接话 · ${counters.share}/2分享 · ${counters.balanced}/1完整结构` },
      common: { met: topicMax >= 3 && counters.relevant >= 2, current: `同一话题深度 ${topicMax}/3 · 有效接话 ${counters.relevant}/2` },
      askedBack: { met: counters.herQuestions >= 1 && counters.share >= 1, current: `她主动反问 ${counters.herQuestions}/1 · 自我分享 ${counters.share}/1` },
      naturalClose: { met: counters.ending >= 1 && currentSession.turn >= 3 && counters.noPressure >= 2, current: `交流 ${currentSession.turn}/3轮 · 自然收尾 ${counters.ending}/1` },
      invite: { met: counters.concreteInvite >= 1 && counters.acceptUncertainty >= 1, current: `具体邀约 ${counters.concreteInvite}/1 · 留有空间 ${counters.acceptUncertainty}/1` },
      boundary: { met: (counters.acceptUncertainty >= 1 || counters.ending >= 1 || counters.respectRejection >= 1) && counters.noPressure >= 1, current: `降速/收尾 ${counters.acceptUncertainty + counters.ending + counters.respectRejection}/1 · 不催促 ${counters.noPressure}/1` }
    };
    return { goal, ...(checks[goal] || checks.balanced) };
  }

  function renderFreeGoalProgress(currentSession) {
    const status = getFreeGoalStatus(currentSession);
    const goal = FREE_PRACTICE_OPTIONS.goals[status.goal];
    return `<div class="free-goal-progress ${status.met ? "met" : ""}"><strong>${status.met ? "目标条件已满足" : goal.label}</strong><span>${escapeHtml(status.current)}</span></div>`;
  }

  function renderFeedback(feedback) {
    const scoreClass = feedback.score >= 78 ? "score-good" : feedback.score >= 60 ? "score-mid" : "score-bad";
    const good = feedback.strengths.length ? feedback.strengths.map(item => `<li>${escapeHtml(item)}</li>`).join("") : "<li>这一句暂时没有明显加分点。</li>";
    const issues = feedback.issues.length ? feedback.issues.map(item => `<li>${escapeHtml(item.text)}</li>`).join("") : "<li>没有发现明显问题。</li>";
    const intentHtml = feedback.intents?.length
      ? `<div class="intent-row">${feedback.intents.map(item => `<span class="intent-chip">${escapeHtml(item)}</span>`).join("")}</div>`
      : "";
    return `
      <div class="score-badge ${scoreClass}">本句 ${feedback.score} 分 · ${escapeHtml(feedback.grade)}</div>
      ${intentHtml}
      <div class="feedback-section"><strong>做得好的地方</strong><ul class="feedback-list">${good}</ul></div>
      <div class="feedback-section"><strong>需要调整</strong><ul class="feedback-list">${issues}</ul></div>
      <div class="feedback-section"><strong>她可能的感受</strong><p>${escapeHtml(feedback.feeling)}</p></div>
      <div class="feedback-section"><strong>她为什么这样回复</strong><p><span class="state-badge">${escapeHtml(feedback.stateLabel || "正常观察")}</span> ${escapeHtml(feedback.reaction || "她会根据上下文继续回应。")}</p></div>
      <div class="feedback-section"><strong>更自然的方向</strong><div class="reference-box">${escapeHtml(feedback.reference)}</div></div>

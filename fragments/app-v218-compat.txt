  // V2.18 连续对话兼容层：保留原规则引擎，仅修复误判与状态计算。
  const analyzeMessageV217 = analyzeMessage;
  analyzeMessage = function analyzeMessageV218(text, currentSession) {
    const analysis = analyzeMessageV217(text, currentSession);
    const normalized = analysis.text || String(text || "").trim();
    const explicitProfileReference =
      /(头像|主页|资料(?:里|上)?|背景图)/.test(normalized) ||
      /你(?:这|那|的)?(?:张)?(?:照片|相片)/.test(normalized) ||
      /(?:照片|相片).{0,10}(?:哪里|哪儿|哪拍|拍的|西湖|景色|风景)/.test(normalized);
    const profileStageAllowed = currentSession.turn <= 1 && currentSession.level.chapter <= 2;

    if (analysis.flags.profileReference && (!explicitProfileReference || !profileStageAllowed)) {
      analysis.flags.profileReference = false;
      analysis.intents = (analysis.intents || []).filter(item => item !== "资料切入");
      analysis.strengths = (analysis.strengths || []).filter(item => !item.includes("资料或照片"));
      analysis.score = clamp(analysis.score - (currentSession.level.chapter <= 2 ? 10 : 5), 0, 100);
    }

    const recentHerText = currentSession.messages
      .filter(message => message.from === "her")
      .slice(-3)
      .map(message => message.text)
      .join(" ");
    const recentTopics = new Set(detectTopicsFromText(recentHerText));
    const carriedTopics = detectTopicsFromText(normalized).filter(topic => recentTopics.has(topic));

    if (carriedTopics.length) {
      analysis.context ||= { topicMatch: [], lastContext: analyzeContext("") };
      analysis.context.topicMatch = [...new Set([...(analysis.context.topicMatch || []), ...carriedTopics])];
      if (!analysis.relevant) {
        analysis.relevant = true;
        analysis.score = clamp(analysis.score + 10, 0, 100);
        if (!analysis.intents.includes("接住话题")) analysis.intents.push("接住话题");
        if (!analysis.strengths.includes("结合了最近几轮内容，话题衔接自然。")) {
          analysis.strengths.push("结合了最近几轮内容，话题衔接自然。");
        }
        analysis.issues = (analysis.issues || []).filter(issue => !["topicJump", "generic"].includes(issue.key));
      }
    }

    return analysis;
  };

  const updateSessionMetricsV217 = updateSessionMetrics;
  updateSessionMetrics = function updateSessionMetricsV218(analysis, currentSession = session) {
    if (!currentSession) throw new Error("缺少训练会话，无法更新状态");
    const pressureBefore = currentSession.metrics.pressure;
    const result = updateSessionMetricsV217(analysis, currentSession);
    const severe = analysis.flags.sexual || analysis.flags.insult || analysis.flags.pressure ||
      (REJECTION_LEVEL_IDS.includes(currentSession.level.id) && !analysis.flags.respectRejection);
    const pressureDelta = severe
      ? 22
      : analysis.score < 45
        ? 10
        : (analysis.score < 60 || analysis.issues.length >= 2)
          ? 5
          : analysis.score >= 75
            ? -4
            : -1;
    currentSession.metrics.pressure = clamp(pressureBefore + pressureDelta, 0, 100);
    return result;
  };

  function getPrimaryIssueV221(analysis) {
    const issues = analysis?.issues || [];
    const priority = ["disrespect","pressure","mindReading","scorekeeping","gamePlaying","overIntimacy","interrogation","scenarioPace","adviceRush","noShare","topicJump","genericOpener","coldReply","vagueInvite","overExplain","tooLong","tooShort"];
    return priority.find(key => issues.some(item => item.key === key)) || issues[0]?.key || null;
  }

  function getAlternativeV221(analysis, completed) {
    const sceneAlternative = getSceneAlternativeV221(completed);
    const issueKey = getPrimaryIssueV221(analysis);
    if (issueKey && REPLAY_ISSUE_COPY_V221[issueKey]) return REPLAY_ISSUE_COPY_V221[issueKey].alternative;
    if (analysis?.score >= 78) return "这句结构可以保留：回应具体内容、补充一点自己，并只留下一个容易回答的问题。";
    if (sceneAlternative) return sceneAlternative;
    if (analysis?.flags?.ending) return "你先忙，之后有空再聊。";
    if (analysis?.flags?.empathy) return "听起来确实不容易。你现在更想让我听你说，还是一起想办法？";
    return "先回应她刚才的具体内容，再补充一点自己的信息，最后只留一个问题。";
  }

  function getNextStepV221(analysis) {
    const issueKey = getPrimaryIssueV221(analysis);
    if (issueKey && REPLAY_ISSUE_COPY_V221[issueKey]) return REPLAY_ISSUE_COPY_V221[issueKey].next;
    if (analysis?.score >= 78) return "保留这句的交换感，不需要为了显得厉害而继续加内容。";
    return "下一句先接住对方，再决定是否分享或提问。";
  }

  function findTurningPointsV221(analyses) {
    const points = new Map();
    if (!analyses.length) return points;
    const add = (index, reason) => {
      if (index < 0 || index >= analyses.length) return;
      const list = points.get(index) || [];
      if (!list.includes(reason)) list.push(reason);
      points.set(index, list);
    };
    const best = analyses.reduce((bestIndex, item, index, list) => item.score > list[bestIndex].score ? index : bestIndex, 0);
    const worst = analyses.reduce((worstIndex, item, index, list) => item.score < list[worstIndex].score ? index : worstIndex, 0);
    add(best, "本局最佳表达");
    add(worst, "本局最低分句");
    const boundary = analyses.findIndex(item => item.flags?.respectRejection || item.flags?.selfBoundary || item.flags?.ending || item.flags?.acceptUncertainty);
    if (boundary >= 0) add(boundary, "出现关键边界处理");
    const risk = analyses.findIndex(item => item.flags?.pressure || item.flags?.mindReading || item.flags?.scorekeeping || item.flags?.intimacy || item.score < 45);
    if (risk >= 0) add(risk, "关系压力明显上升");
    let upIndex = -1, upDelta = 0, downIndex = -1, downDelta = 0;
    for (let index = 1; index < analyses.length; index += 1) {
      const delta = analyses[index].score - analyses[index - 1].score;
      if (delta > upDelta) { upDelta = delta; upIndex = index; }
      if (delta < downDelta) { downDelta = delta; downIndex = index; }
    }
    if (upDelta >= 12) add(upIndex, `比上一句提升${upDelta}分`);
    if (downDelta <= -12) add(downIndex, `比上一句下降${Math.abs(downDelta)}分`);
    return points;
  }

  function buildSessionReplayV221(completed, result = {}) {
    const analyses = completed?.analyses || [];
    const myMessages = (completed?.messages || []).filter(item => item.from === "me");
    const herMessages = (completed?.messages || []).filter(item => item.from === "her");
    const turningPoints = findTurningPointsV221(analyses);
    const issueCounts = {};
    analyses.forEach(item => (item.issues || []).forEach(issue => { issueCounts[issue.key] = (issueCounts[issue.key] || 0) + 1; }));
    const turns = analyses.map((analysis, index) => {
      const issueKey = getPrimaryIssueV221(analysis);
      const severe = analysis.flags?.pressure || analysis.flags?.insult || analysis.flags?.sexual || analysis.flags?.mindReading || analysis.flags?.scorekeeping;
      const tone = severe || analysis.score < 55 ? "bad" : analysis.score < 78 ? "mid" : "good";
      return {
        index,
        turn: index + 1,
        userText: String(myMessages[index]?.text || analysis.text || "").slice(0, 240),
        replyText: String(herMessages[index + 1]?.text || "").slice(0, 240),
        score: Number(analysis.score || 0),
        tone,
        label: tone === "good" ? "自然有效" : tone === "mid" ? "可继续但能更好" : "需要调整",
        keyReasons: turningPoints.get(index) || [],
        issueKey,
        issueLabel: issueKey ? (REPLAY_ISSUE_COPY_V221[issueKey]?.label || ISSUE_LABELS[issueKey] || issueKey) : null,
        issueText: analysis.issues?.[0]?.text || null,
        strength: analysis.strengths?.[0] || null,
        semantic: analysis.semantic ? { action: analysis.semantic.action, object: analysis.semantic.object, context: analysis.semantic.context } : null,
        alternative: getAlternativeV221(analysis, completed),
        nextStep: getNextStepV221(analysis),
        flags: {
          questionCount: Number(analysis.questionCount || 0),
          pressure: Boolean(analysis.flags?.pressure),
          boundary: Boolean(analysis.flags?.respectRejection || analysis.flags?.selfBoundary || analysis.flags?.ending || analysis.flags?.acceptUncertainty)
        }
      };
    });
    const best = turns.length ? turns.reduce((a,b) => b.score > a.score ? b : a) : null;
    const worst = turns.length ? turns.reduce((a,b) => b.score < a.score ? b : a) : null;
    const topIssue = Object.entries(issueCounts).sort((a,b) => b[1]-a[1])[0] || null;
    const ineffectiveQuestions = turns.filter(turn => turn.flags.questionCount >= 2 || ["interrogation","noShare"].includes(turn.issueKey)).length;
    const boundarySignals = turns.filter(turn => turn.flags.boundary).length;
    const nextFocus = topIssue
      ? (REPLAY_ISSUE_COPY_V221[topIssue[0]]?.next || ISSUE_COACHING[topIssue[0]] || "下一局先修正出现次数最多的问题。")
      : "下一局提高难度，继续保持回应、分享和留空间。";
    return {
      replayVersion: REPLAY_VERSION_V221,
      createdAt: new Date().toISOString(),
      title: completed?.level?.title || "训练复盘",
      objective: completed?.level?.objective || "",
      characterId: completed?.char?.id || "linxia",
      characterName: completed?.char?.name || "角色",
      personality: completed?.personality?.label || "独立性格",
      trainingMode: completed?.trainingMode || "coach",
      type: completed?.realScenario ? "scenario" : completed?.isMistake ? "mistake" : completed?.isFree ? "free" : "formal",
      scenarioId: completed?.realScenario?.id || null,
      passed: Boolean(result.passed),
      avg: Number(result.avg || 0),
      finalState: getVisibleCharacterState(completed),
      bestTurn: best?.turn || 0,
      worstTurn: worst?.turn || 0,
      ineffectiveQuestions,
      boundarySignals,
      topIssue: topIssue ? { key: topIssue[0], count: topIssue[1], label: REPLAY_ISSUE_COPY_V221[topIssue[0]]?.label || ISSUE_LABELS[topIssue[0]] || topIssue[0] } : null,
      nextFocus,
      turns
    };
  }

  function openReplayV221(replay) {
    if (!replay?.turns) return false;
    const dialog = ensureReplayDialogV221();
    const content = document.getElementById("v221ReplayContent");
    if (content) content.innerHTML = renderReplayHtmlV221(replay);
    try { dialog.showModal(); } catch {}
    return true;
  }
